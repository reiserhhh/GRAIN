\[
K_x(a)=\exp[-\|x-a\|^2/(2s^2)],\qquad s^2=.05525111588037168.

\[
\begin{aligned}
F_g(x,z)&=E_{i\mid g}E_t[K_x(a_{it})K_z(b_{it})],\\
J_g(x,z)&=E_{i\mid g}[E_tK_x(a_{it})E_tK_z(b_{it})],\\
I_g(x,z)&=E_{i\mid g}E_tK_x(a_{it})\ E_{j\mid g}E_sK_z(b_{js}).

\[
